'number' + 3 + 3   - > "number33" // 3 до рядка JS додасть просто як символ, а не як цифру, конкатенація 

null + 3  - > 3  // JS перетворить null на 0, щоб виконати арифметичну операцію потрібні однакові типи даних, числа

5 && "qwerty"  - > qwerty // && не звичайне і, шукає першу правдиву умову, переходить до другої, якщо знайшов першу

+'40' + +'2' + "hillel"  - > "42hillel" // 1й і 3й + перетворюють рядок на число, 2й + додає числа, 4й + об'єднує з рядком

'10' - 5 === 6  - > false // 5 не дорівнює 6, 10 перетворить з рядка на число

true + false  - > 1 // true це 1, false це 0

'4px' - 3  - > NaN // не зможе виконати дію з рядком, результат не число

'4' - 3  - > 1 // змінить 4 з рядка на число, щоб провести арифметичну операцію

'6' + 3 ** 0  - > 61 // ** буде першої дією, результат 1, треба скласти рядок і число, конкатенація 

12 / '6'  - > 2  // змінить 6 з рядка на число, отримає однакові типи, проведе арифметичну операцію

'10' + (5 === 6)  - > "10false" // 5 не дорівнює 6 тому false, потім конкатенація 

null == ''  - > true // null - це як порожнеча, рядок не заповнений, == не перевіряє типи

3 ** (9 / 3)  - > 27 // спочатку (), потім піднесення до степеня 

!!'false' == !!'true'  - > true  // заповнений рядок це true, !! заперечення, після якого знову заперечення, істина порівнювалась з істиною 

0 || '0' && 1  - > 1 // рядок заповнений, тому true, true це 1, або шукає перше значення, що підходить умові, 0 не підходить, 1 підходить

(+null == false) < 1  - > false // унарний +, дає число 0, несурова рівність вірна, це true, тобто 1

false && true || true  - > true // і зупинитися на false, умова не виконується, або шукає до першої правдивої умови, це буде останнє true

false && (false || true)  - > false // () перші, маємо true, але хиба і правда, умова від початку не виконалася для логічного і, то маємо false

(+null == false) < 1 ** 5  - > false // унарний +, маємо 0, рівність вірна, це true, тобто 1, 1 не менше за 1 (** це степень)
